document.addEventListener('DOMContentLoaded', function() {
    // Select all links with the class "link_with_param"
    const links = document.querySelectorAll('.link_with_param');
    const currentUrl = window.location.href;

    // Check if there are query parameters in the current URL
    const queryString = currentUrl.includes('?') ? currentUrl.substring(currentUrl.indexOf('?')) : '';

    // Update each link's href to include query parameters
    links.forEach(link => {
        link.href += queryString;
    });
});

function is_validateDate(year, month, day) {

	day = Number(day);
	month = Number(month) - 1; //bloody 0-indexed month
	year = Number(year);

	let d = new Date(year, month, day);

	if (new Date().getTime() - d.getTime() < 0) {
		return false;
	}

	let yearMatches = d.getFullYear() === year;
	let monthMatches = d.getMonth() === month;
	let dayMatches = d.getDate() === day;

	return yearMatches && monthMatches && dayMatches;
}

function IsHKID(str) {
	var strValidChars = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
	if (str.length < 8) {
		return false;
	}
	str = str.toUpperCase();
	var hkidPat = /^([A-Z]{1,2})([0-9]{6})([A0-9])$/;
	var matchArray = str.match(hkidPat);
	if (matchArray == null) {
		return false;
	}
	var charPart = matchArray[1];
	var numPart = matchArray[2];
	var checkDigit = matchArray[3];
	var checkSum = 0;
	if (charPart.length == 2) {
		checkSum += 9 * (10 + strValidChars.indexOf(charPart.charAt(0)));
		checkSum += 8 * (10 + strValidChars.indexOf(charPart.charAt(1)));
	} else {
		checkSum += 9 * 36;
		checkSum += 8 * (10 + strValidChars.indexOf(charPart));
	}

	for (var i = 0, j = 7; i < numPart.length; i++, j--) {
		checkSum += j * numPart.charAt(i);
	}
	var remaining = checkSum % 11;
	var verify = remaining == 0 ? 0 : 11 - remaining;
	return verify == checkDigit || (verify == 10 && checkDigit == 'A');
}

function hkid_display(str) {
	return str.slice(0, str.length - 1) + "(" + str[str.length - 1] + ")";
}

function ec_display(str) {
	return str.slice(0, 4) + "-" + str.slice(4, 11) + "-" + str.slice(11, 13) + "(" + str[str.length - 1] + ")";
}

	var birthday_input_formatted = birthday_input.replace(/-/g, "/");
	birthday_input_formatted = birthday_input_formatted.replace(/\/00/g, "/01");
	birthday = new Date(birthday_input_formatted);
	var year = birthday.getFullYear();
	var month = birthday.getMonth();
	var day = birthday.getDate();

	if (isNaN(month)) {
		year = birthday_input.substring(0, birthday_input.indexOf('-'));
		month = 0;
		day = 1;
	}
	return new Date(year, month, day);
}

	cleanPMIDataAndCVDBLastDoseHistory();

		success: function(returnData) {

					}

					}

						}

				}else{

				}

				if(typeof response.result !== 'undefined' && response.result=="Unmatched"){
					handleEHRFail("expiredTab");

			//show_error_page(7,xhr.status,'');

		}
	});

		var isFromEHR = false;

			(isContainOTPandAppsource())){
			const elements = document.querySelectorAll('.ehrOptOutRemove');
	
			elements.forEach(element => {
			    element.remove();
			});
			isFromEHR = true;
		}
		
		var errorCode = "";
		if(xhr != "" && xhr!=null && xhr.responseJSON!=null && xhr.responseJSON.error!=null){
			errorCode = xhr.responseJSON.error;

					showPopUp(getString("ehr_popup_dob_expired_tab_title", lang),getString("ehr_popup_dob_expired_tab_content", lang), getString("display_btn_CLOSE", lang));
				}
			}
		} else if (isCurrentC19()) {

				if(isFromEHR && errorCode.includes("docType")){
					showPopUp(getString("c19_ehr_popup_doc_type_not_support_title", lang), getString("c19_ehr_popup_doc_type_not_support_content", lang), getString("display_btn_CLOSE", lang));

function prefillPMIData(userProfileResponse, isDisabled) {
	
	if(userProfileResponse == null){
		return;
	}
	
	//dateOfBirth of exactDobFlag (EDMY/EMY/EY)
	if(userProfileResponse.exactDobFlag && userProfileResponse.dateOfBirth) {
		
		const dobParts = userProfileResponse.dateOfBirth.split('-');
		const exactDobFlagTrimmed = userProfileResponse.exactDobFlag.trim();
		
		if(exactDobFlagTrimmed == "EDMY"){
			//EDMY: Day + Month + Year
			const day = dobParts[0];
		    const month = dobParts[1];
		    const year = dobParts[2];
		    
		    var twoDigitsMonth = convertMonthAbbreviationToTwoDigits(month);
		    
			$('#step_1_date_of_birth_day').val(day).change();
			$('#step_1_date_of_birth_month').val(twoDigitsMonth).change();
			$('#step_1_date_of_birth_year').val(year).change();

		} else if(exactDobFlagTrimmed == "EMY") {
			//EMY: Month + Year, Day 'Not Applicable'
		    const month = dobParts[0];
		    const year = dobParts[1];
		    
			var twoDigitsMonth = convertMonthAbbreviationToTwoDigits(month);
			
			$('#step_1_date_of_birth_day').val('-').change();
			$('#step_1_date_of_birth_month').val(twoDigitsMonth).change();
			$('#step_1_date_of_birth_year').val(year).change();
			
		} else if(exactDobFlagTrimmed == "EY") {
			//EY: Year, Month & Day 'Not Applicable'
			const year = dobParts[0];
			
			$('#step_1_date_of_birth_day').val('-').change();
			$('#step_1_date_of_birth_month').val('-').change();
			$('#step_1_date_of_birth_year').val(year).change();
		} 
		
		if(isDisabled){
			$('#step_1_date_of_birth_day').prop('disabled', true);
			$('#step_1_date_of_birth_month').prop('disabled', true);
			$('#step_1_date_of_birth_year').prop('disabled', true);
		}
	}
	
    if(isValueValid(userProfileResponse.idNo) && userProfileResponse.idNo.hkid){
	    
		$('#step_1_ATGC_NATCARDTYPE').val(getPMIDataDocType_Detail(userProfileResponse)).change();
		$('#step_1_documentId_HKIC_prefix').val(userProfileResponse.idNo.hkid.slice(0, -1)).change();
		$('#step_1_documentId_HKIC_check_digit').val(userProfileResponse.idNo.hkid.slice(-1)).change();

		if (isDisabled) {
			$('#step_1_ATGC_NATCARDTYPE').val(getPMIDataDocType_Detail(userProfileResponse)).addClass('disabled-container');
			$('#step_1_documentId_HKIC_prefix').addClass('disabled-container');
			$('#step_1_documentId_HKIC_check_digit').addClass('disabled-container');
			$('.step_1_documentId_detail .open_parenthesis').addClass('disabled-container');
			$('.step_1_documentId_detail .close_parenthesis').addClass('disabled-container');
			$('.step_1_documentId_detail').addClass('disabled-container');
			$('#step_1_ATGC_NATCARDTYPE').prop('disabled', true);
		}
	    
	}

	//Step2 start 
	$('#step_2_last_name').val(userProfileResponse.englishSurname).change();
    $('#step_2_first_name').val(userProfileResponse.englishGivenName).change();
	if(isDisabled){
		$('#step_2_last_name').addClass('disabled-container');
    	$('#step_2_first_name').addClass('disabled-container');
	}
	
    if(isValueValid(userProfileResponse.mobile)){
		$('#step_2_tel_for_sms_notif').val(userProfileResponse.mobile.mobilePhone).change();
	    $('#step_2_tel_for_sms_notif_confirm').val(userProfileResponse.mobile.mobilePhone).change();

		//requirement changed, mobile can update even prefilled
		/*if(isDisabled){
		    $('#step_2_tel_for_sms_notif').addClass('disabled-container');
	    	$('#step_2_tel_for_sms_notif_confirm').addClass('disabled-container');
		}*/
	}
	
	if(isValueValid(userProfileResponse.ehrNo)){
		$('#step_2_ehr_number_prefilled').val(userProfileResponse.ehrNo);
	}
	
    const gender = userProfileResponse.sex;
    if(gender){
	    if (gender === 'M') {
	        $('#step_2_gender_male').prop('checked', true).change();
	    } else if (gender === 'F') {
	        $('#step_2_gender_female').prop('checked', true).change();
	    } else {
	        $('#step_2_gender_other').prop('checked', true).change();
	    }
	    
	    $('#step_2_gender_prefilled').val(gender);
	    
	    if(isDisabled){
		    $('#radio_group_gender input[type="radio"]').prop('disabled', true);
	    	$('#radio_group_gender label').addClass('disabled-label');
		}
	    
	}

}

			//EDMY: Day + Month + Year
			const day = dobParts[0];
		    const month = dobParts[1];
		    const year = dobParts[2];
		    
		    var twoDigitsMonth = convertMonthAbbreviationToTwoDigits(month);
		    
			$('#step_1_date_of_birth_day').val(day).change();
			$('#step_1_date_of_birth_month').val(twoDigitsMonth).change();
			$('#step_1_date_of_birth_year').val(year).change();

		} else if(exactDobFlagTrimmed == "EMY") {
			//EMY: Month + Year, Day 'Not Applicable'
		    const month = dobParts[0];
		    const year = dobParts[1];
		    
			var twoDigitsMonth = convertMonthAbbreviationToTwoDigits(month);
			
			$('#step_1_date_of_birth_day').val('-').change();
			$('#step_1_date_of_birth_month').val(twoDigitsMonth).change();
			$('#step_1_date_of_birth_year').val(year).change();
			
		} else if(exactDobFlagTrimmed == "EY") {
			//EY: Year, Month & Day 'Not Applicable'
			const year = dobParts[0];
			
			$('#step_1_date_of_birth_day').val('-').change();
			$('#step_1_date_of_birth_month').val('-').change();
			$('#step_1_date_of_birth_year').val(year).change();
		} 
		
		if(isDisableAfterPrefillPMIData){
			$('#step_1_date_of_birth_day').prop('disabled', true);
			$('#step_1_date_of_birth_month').prop('disabled', true);
			$('#step_1_date_of_birth_year').prop('disabled', true);
		}
	}
	
    if(isValueValid(userProfileResponse.idNo) && userProfileResponse.idNo.hkid){
		
		$('#step_1_ATGC_NATCARDTYPE').val(getPMIDataDocType_Detail(userProfileResponse)).change();
		if(hkidFormatList.includes(getPMIDataDocType_Detail(userProfileResponse))){
			$('#step_1_documentId_HKIC_prefix').val(userProfileResponse.idNo.hkid.slice(0, -1)).change();
			$('#step_1_documentId_HKIC_check_digit').val(userProfileResponse.idNo.hkid.slice(-1)).change();
		}else{
			// for Certificate of exemption
		}
		
		if (isDisableAfterPrefillPMIData) {
			$('#step_1_ATGC_NATCARDTYPE').val(getPMIDataDocType_Detail(userProfileResponse)).addClass('disabled-container');
			$('#step_1_documentId_HKIC_prefix').addClass('disabled-container');
			$('#step_1_documentId_HKIC_check_digit').addClass('disabled-container');
			$('.step_1_documentId_detail .open_parenthesis').addClass('disabled-container');
			$('.step_1_documentId_detail .close_parenthesis').addClass('disabled-container');
			$('.step_1_documentId_detail').addClass('disabled-container');
			$('#step_1_ATGC_NATCARDTYPE').prop('disabled', true);
		}
		
	}
	
	let isEmptyOrWrongDoseHistory = true;
	if(cvdbLastDoseHistory != null && cvdbLastDoseHistory.length == 1){
		isEmptyOrWrongDoseHistory = false;
		
		//set target dose round   supportedDoseRound
		if(cvdbLastDoseHistory[0].doseSeqNo !== null 
				&& cvdbLastDoseHistory[0].doseSeqNo > 0 
				&& cvdbLastDoseHistory[0].doseSeqNo < supportedDoseRound){
			let targetDoseRound = (parseInt(cvdbLastDoseHistory[0].doseSeqNo) || 0) + 1;
			$("#select_additional_dose").val(targetDoseRound).change();
			if(isDisableAfterPrefillPMIData){
				$('#select_additional_dose').prop('disabled', true);
			}
		}
		
		
		//set dose history date
		if(cvdbLastDoseHistory[0].doseDate !== null){
			let lastDoseHistoryDate = parseYYYYMMDD(cvdbLastDoseHistory[0].doseDate);
			if(lastDoseHistoryDate !== null){
				$('#step_1_date_of_second_dose_day').val(lastDoseHistoryDate.day).change();
				$('#step_1_date_of_second_dose_month').val(lastDoseHistoryDate.month).change();
				$('#step_1_date_of_second_dose_year').val(lastDoseHistoryDate.year).change();
				
				if(isDisableAfterPrefillPMIData){
					$('#step_1_date_of_second_dose_day').prop('disabled', true);
					$('#step_1_date_of_second_dose_month').prop('disabled', true);
					$('#step_1_date_of_second_dose_year').prop('disabled', true);
				}
			} 
		}
		
		
		
	}
	if(isEmptyOrWrongDoseHistory){
		$("#select_additional_dose").val("1").change();
		$('#select_additional_dose').prop('disabled', true);
	}

    
    
    
    //Step2 start 
   	$('#step_2_name').val(userProfileResponse.englishSurname + " " + userProfileResponse.englishGivenName).change();
	if(isDisableAfterPrefillPMIData){
		$('#step_2_name').addClass('disabled-container');
	}
	
	if(isValueValid(userProfileResponse.ehrNo)){
		$('#step_2_ehr_number_prefilled').val(userProfileResponse.ehrNo);
	}
	
    if(isValueValid(userProfileResponse.mobile)){
		$('#step_2_tel_for_sms_notif').val(userProfileResponse.mobile.mobilePhone).change();
	    $('#step_2_tel_for_sms_notif_confirm').val(userProfileResponse.mobile.mobilePhone).change();
	}
}

// Checks if all date elements are filled. Returns true if valid, false if empty/null.
function dateChecker(year_field, month_field, day_field) {
	var year = $("#" + year_field).val();
	var month = $("#" + month_field).val();
	var day = $("#" + day_field).val();

	if (!year || !month || !day) {
		return false;
	}
	return true;
}

// replace "-" to "01", and return full date with yyyy/mm/dd
function dateFormatter(year_field, month_field, day_field) {
	var year = $("#" + year_field).val();
	var month = $("#" + month_field).val() == "-" ? "01" : $("#" + month_field).val();
	var day = $("#" + day_field).val() == "-" ? "01" : $("#" + day_field).val();
	var full_date = year + "/" + month + "/" + day;
	
	return {
		year: year,
		month: month,
		day: day,
		full_date : full_date
	};
}

function doseSuspendByAgeAndDate(dobObject) {
	const { year, month, day } = dobObject;

	const {
		C19_CHILD_PAUSE_AGE,
		C19_ADULT_PAUSE_AGE,
		C19_CHILD_PAUSE_DATE_TIME,
		C19_ADULT_PAUSE_DATE_TIME
	} = systemParamResponseJs;

	const [childMinAge, childMaxAge] = C19_CHILD_PAUSE_AGE.split("-").map(num => parseInt(num, 10));
	const [adultMinAge, adultMaxAge] = C19_ADULT_PAUSE_AGE.split("-").map(num => parseInt(num, 10));

	const currentAge = getAgeByDobJs(year, month, day);

	let isChild = false;
	let targetCompareDateTimeStr = "";

	if (currentAge >= childMinAge && currentAge <= childMaxAge) {
		isChild = true;
		targetCompareDateTimeStr = C19_CHILD_PAUSE_DATE_TIME;
	} else if (currentAge >= adultMinAge && currentAge <= adultMaxAge) {
		isChild = false;
		targetCompareDateTimeStr = C19_ADULT_PAUSE_DATE_TIME;
	} else {
		// Fallback for ages outside specified configurations
		return { result: false, isChild: false };
	}

	const targetCompareDateTime = moment(targetCompareDateTimeStr, "YYYY-MM-DD HH:mm:ss");
	const isDoseSuspendedByDate = moment().isAfter(targetCompareDateTime);

	return { result: isDoseSuspendedByDate, isChild: isChild };
}

function executeInfectedPersonLogic(value) {
	// Reset inputs and visibility
	$('.step_1_options').prop('checked', false);
	$('.step_1_third_dose').show();
	$("#d4_subtitle_without_infected").hide();
	$("#d4_subtitle_with_infected").hide();

	const isInfected = value === 'I';
	const isNonInfected = value === 'N';
	const isValid = isInfected || isNonInfected;

	$('.dose_priority_group').toggle(isNonInfected);
	$('.dose_priority_group.infected').toggle(isInfected);
	// dose [4] and above is using mix layout
	$('.dose_priority_group.mix').toggle(isValid && selectedDoseRound >= 4);
	
	if (isInfected) {
		$("#d4_subtitle_without_infected").hide();
		$("#d4_subtitle_with_infected").show();
	} else if (isNonInfected) {
		$("#d4_subtitle_without_infected").show();
		$("#d4_subtitle_with_infected").hide();
	}

	const nonInfectedClass = `.d${selectedDoseRound}_elig_grp_non_infected`;
	const infectedClass = `.d${selectedDoseRound}_elig_grp_infected`;

	$(nonInfectedClass).toggle(isNonInfected && selectedDoseRound == 4);
	$(infectedClass).toggle(isValid && selectedDoseRound == 4);

	if (isValid) {
		DR_DisplayOptionByInfectionType(value);
	}
}
